 1. Mở URL:

  https://storage.y8.com/y8-studio/html5/toni/myparkinglot_v3/?key=y8&value=default

  2. Đợi game tải xong.
  3. Nhấn F12 → tab Console.
  4. Dán script dưới đây, nhấn Enter:

  (() => {
    window.__carSpeed10x?.stop?.();

    const state = {
      factor: 10,
      patched: new Set(),
      timer: null,
    };

    function patchCar(car) {
      if (!car || state.patched.has(car)) return;

      let rawSpeed = Number(car.moveSpeed);
      if (!Number.isFinite(rawSpeed)) return;

      Object.defineProperty(car, "moveSpeed", {
        configurable: true,
        enumerable: true,
        get: () => rawSpeed * state.factor,
        set: value => {
          rawSpeed = Number(value);
        },
      });

      state.patched.add(car);
    }

    function scanCars() {
      const scene = window.cc?.director?.getScene?.();
      if (!scene) return;

      function walk(node) {
        for (const component of node._components || []) {
          if (cc.js.getClassName(component.constructor) === "LY07_PTG_Car") {
            patchCar(component);
          }
        }

        for (const child of node.children || []) {
          walk(child);
        }
      }

      walk(scene);
    }

    state.stop = () => {
      clearInterval(state.timer);

      for (const car of state.patched) {
        if (!cc.isValid(car)) continue;

        const originalSpeed = car.moveSpeed / state.factor;
        delete car.moveSpeed;
        car.moveSpeed = originalSpeed;
      }

      state.patched.clear();
      delete window.__carSpeed10x;
    };

    scanCars();
    state.timer = setInterval(scanCars, 250);
    window.__carSpeed10x = state;

    console.log(`Đã tăng ${state.patched.size} xe lên ${state.factor}x`);
  })();

  Đổi thành 5x:

  window.__carSpeed10x.factor = 5;

  Khôi phục tốc độ:

  window.__carSpeed10x.stop();

  Nếu Chrome chặn dán vào Console, gõ thủ công allow pasting, nhấn Enter, rồi dán lại. Reload trang sẽ mất patch.
